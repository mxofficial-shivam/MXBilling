function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Fetch Clients (Sheet tab: Clients -> Col A: Name, Col B: Phone, Col C: Previous Due)
  var clientSheet = ss.getSheetByName("Clients");
  var clientData = clientSheet ? clientSheet.getDataRange().getValues() : [];
  var clients = [];
  for (var i = 1; i < clientData.length; i++) {
    if(clientData[i][0]) {
      clients.push({ name: clientData[i][0], phone: clientData[i][1], previousDue: clientData[i][2] || 0 });
    }
  }

  // Fetch Products (Sheet tab: Products -> Col A: Item Name, Col B: Rate)
  var productSheet = ss.getSheetByName("Products");
  var productData = productSheet ? productSheet.getDataRange().getValues() : [];
  var products = [];
  for (var j = 1; j < productData.length; j++) {
    if(productData[j][0]) {
      products.push({ name: productData[j][0], rate: productData[j][1] || 0 });
    }
  }

  var result = { clients: clients, products: products };
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Invoices") || ss.getActiveSheet();
  var data = JSON.parse(e.postData.contents);

  var itemsList = data.items.map(function(i){ return i.name + " (" + i.qty + "x" + i.rate + ")"; }).join(", ");

  sheet.appendRow([
    new Date(),
    data.clientName,
    data.clientPhone,
    itemsList,
    data.subtotal,
    data.previousDue,
    data.totalPayable,
    data.amountPaid,
    data.balanceDue,
    data.status
  ]);

  return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
}
